  function useCountUp(target) {
    const [v, setV] = React.useState(target); const ref = React.useRef(target);

      raf = requestAnimationFrame(tick); return () => cancelAnimationFrame(raf);
    }, [target]); return v;
  }

  function CompBars({ by_type, ghost }) {

  /* ---- derived deck reads ------------------------------------------------ */
  const ORDER = ["0", "1", "2", "3", "4", "5", "6", "7+"];
  function curveStats(A, deck) {
    const buckets = A.curve;
    const nonland = ORDER.reduce((s, k) => s + (buckets[k] || 0), 0);
    const peakKey = ORDER.reduce((best, k) => ((buckets[k] || 0) > (buckets[best] || 0) ? k : best), "0");
    const low = (buckets["0"] || 0) + (buckets["1"] || 0) + (buckets["2"] || 0) + (buckets["3"] || 0);
    // median MV across nonland spells
    const flat = []; ORDER.forEach((k) => { const v = k === "7+" ? 7 : +k; for (let i = 0; i < (buckets[k] || 0); i++) flat.push(v); });
    const median = flat.length ? flat.sort((a, b) => a - b)[Math.floor(flat.length / 2)] : 0;
    // sample card names per bucket
    const byMv = {}; ORDER.forEach((k) => (byMv[k] = []));
    deck.filter((c) => !isLand(c)).forEach((c) => { const mv = mvOf(c); byMv[mv >= 7 ? "7+" : String(mv)].push(c); });
    return { nonland, peakKey, lowShare: nonland ? low / nonland : 0, median, byMv };
  }

        </div>

        <div className="ic-h">By mana value<span className="ic-h__c">{s.nonland} spells</span></div>
        <div className="ic-rows">
          {ORDER.filter((k) => s.byMv[k].length).map((k) => (
            <div className="ic-row" key={k}>
              <span className="ic-row__k">{k === "7+" ? "7+" : k}</span>
              <span className="ic-row__names">{names(s.byMv[k], 3) || "—"}</span>
              <span className="ic-row__n">{s.byMv[k].length}</span>
            </div>
          ))}
        </div>
        <div className="ic-callout"><Ic.Info /><span>Curve peaks at <b>MV {s.peakKey}</b>; <b>{pct(s.lowShare)}</b> of nonland spells cost ≤ 3 — castable through the early turns.</span></div>
      </div>
    );
  }

  function PipCard({ A, PR }) {
    const total = COLORS.reduce((t, c) => t + (A.pips[c] || 0), 0);
    const present = COLORS.filter((c) => (A.pips[c] || 0) > 0 || (A.sources[c] || 0) > 0);
    const top = present.reduce((b, c) => ((A.pips[c] || 0) > (A.pips[b] || 0) ? c : b), present[0] || "W");
    const max = Math.max(1, ...present.map((c) => Math.max(A.pips[c] || 0, A.sources[c] || 0)));
    const NAME = { W: "white", U: "blue", B: "black", R: "red", G: "green" };
    const tight = present.filter((c) => (A.sources[c] || 0) < (A.pips[c] || 0) * 0.7);
    return (
      <div className="ic-col">
        <div className="ic-eyebrow">Color pips required<span className="ic-eyebrow__c">{total} colored</span></div>
        <Pips pips={A.pips} ghost={PR ? PR.pips : null} />
        <hr className="ic-sep" />
        <div className="ic-h">Demand vs supply</div>
        <div className="ic-dd">
          {present.map((c) => {
            const demand = A.pips[c] || 0, supply = A.sources[c] || 0;
            const tightC = supply < demand * 0.7;
            return (
              <div className="ic-dd__row" key={c}>
                <i className={`ms ms-${c.toLowerCase()} ms-cost`} style={{ fontSize: 16 }} />
                <span className="ic-dd__bars">
                  <span className="ic-dd__bar"><span className="ic-dd__fill ic-dd__fill--demand" style={{ width: (demand / max) * 100 + "%" }} /></span>
                  <span className="ic-dd__bar"><span className={"ic-dd__fill ins-pip__fill--" + c} style={{ width: (supply / max) * 100 + "%" }} /></span>
                </span>
                <span className="ic-dd__meta"><b>{demand}</b> / {supply} <span className={tightC ? "warn" : "ok"}>{tightC ? "tight" : "ok"}</span></span>
              </div>
            );
          })}
        </div>
        <div className="ic-legend"><span><i className="ic-dd__fill--demand" />Pips required</span><span><i style={{ background: "var(--mtg-g)" }} />Sources available</span></div>
        <div className="ic-callout"><Ic.Info /><span>Heaviest in <b>{NAME[top]}</b> ({A.pips[top]} pips). {tight.length ? <span><b>{tight.map((c) => NAME[c]).join(", ")}</b> demand outpaces sources — watch early casts.</span> : <span>Every color is comfortably supported.</span>}</span></div>
      </div>
    );
  }

  function TypeCard({ A, PR, deck }) {
    const t = A.composition;
    const get = (k) => t[k] || 0;
    const perms = get("Creature") + get("Artifact") + get("Enchantment") + get("Planeswalker") + get("Land");
    const spells = get("Instant") + get("Sorcery");
    const lands = A.lands; const nonland = (A.stats.total_cards - 1) - lands;
    const roleC = A.roleCount || {};
    const interaction = (roleC.spot_removal || 0) + (roleC.board_wipe || 0) + (roleC.counterspell || 0);
    const tot = Math.max(1, A.stats.total_cards - 1);
    return (
      <div className="ic-col">
        <div className="ic-eyebrow">Composition<span className="ic-eyebrow__c">{A.stats.total_cards - 1} + commander</span></div>
        <CompBars by_type={t} ghost={PR ? PR.composition : null} />
        <hr className="ic-sep" />
        <div className="ic-statline">
          <Stat k="Permanents" v={perms} sub={pct(perms / tot)} />
          <Stat k="Spells" v={spells} sub={pct(spells / tot)} />
          <Stat k="Lands" v={lands} accent />
          <Stat k="Nonland" v={nonland} />
          <Stat k="Interaction" v={interaction} />
          <Stat k="Walkers" v={get("Planeswalker")} />
        </div>
        <div className="ic-callout"><Ic.Info /><span><b>{pct(perms / tot)}</b> permanents, <b>{interaction}</b> interaction pieces (removal + wipes + counters). {get("Planeswalker")} planeswalkers anchor the superfriends plan.</span></div>
      </div>
    );
  }

  function SourceCard({ A }) {
    const present = COLORS.filter((c) => (A.sources[c] || 0) > 0 || (A.pips[c] || 0) > 0);
    const totalSrc = COLORS.reduce((s, c) => s + (A.sources[c] || 0), 0);
    const under = present.filter((c) => (A.sources[c] || 0) < A.threshold);
    const NAME = { W: "white", U: "blue", B: "black", R: "red", G: "green" };
    return (
      <div className="ic-col">
        <div className="ic-eyebrow"><Ic.Flask />Mana sources<span className="ic-eyebrow__c">{A.lands} lands</span></div>
        <ManaSourceBars sources={A.sources} threshold={A.threshold} underSupported={under} />
        <hr className="ic-sep" />
        <div className="ic-statline">
          <Stat k="Sources" v={totalSrc} sub="total" />
          <Stat k="Threshold" v={A.threshold} accent />
          <Stat k="Colors" v={present.filter((c) => (A.sources[c] || 0) > 0).length} />
          <Stat k="Under" v={under.length} />
        </div>
        <div className="ic-callout"><Ic.Info /><span>Each color wants <b>≥ {A.threshold}</b> sources for reliable on-curve casting. {under.length ? <span><b>{under.map((c) => NAME[c]).join(", ")}</b> sit below threshold — add fixing.</span> : <span>All colors clear the threshold.</span>}</span></div>
      </div>
    );
  }

  function RoleCard({ A, PR, deck }) {
    const cardsFor = (role) => role === "land" ? deck.filter(isLand) : deck.filter((c) => (c.roles || []).includes(role));
    return (
      <div className="ic-col">
        <div className="ic-eyebrow"><Ic.Sparkles />Role coverage<span className="ic-eyebrow__c">have / target band</span></div>
        <RoleCoverageBars gaps={A.coverage} ghost={PR ? PR.coverage : null} />
        <hr className="ic-sep" />
        <div className="ic-rows">
          {A.coverage.map((g) => {
            const cards = cardsFor(g.role);
            const need = g.want_min - g.have, trim = g.have - g.want_max;
            return (
              <div className="ic-role" key={g.role}>
                <div className="ic-role__head">
                  <RoleChip role={g.role} />
                  <span className={"ic-status ic-status--" + g.status}>{g.status}</span>
                  <span className="ic-role__band"><b>{g.have}</b> / {g.want_min}–{g.want_max}</span>
                </div>
                <div className="ic-role__cards">{cards.length ? names(cards, 4) : "—"}</div>
                {g.status === "under" ? <div className="ic-role__fix"><Ic.ArrowUp />Add {need} more to hit the band</div> : g.status === "over" ? <div className="ic-role__fix" style={{ color: "var(--info)" }}><Ic.ArrowDown />{trim} over — consider trimming</div> : null}
              </div>
            );
          })}
        </div>
      </div>
    );
  }

  function BracketCard({ A }) {

      { ok: gc <= 3, t: "Game Changers ≤ 3", d: gc ? P.game_changers.join(", ") : "none" },
      { ok: P.mld.length === 0, t: "No mass land denial", d: P.mld.length ? P.mld.join(", ") : "none present" },
      { ok: P.combos.length === 0, t: "No early 2-card combos", d: P.combos.length ? P.combos.join(", ") : "none detected" },
      { ok: P.fast_mana.length <= 2, t: "Fast mana ≤ 2", d: P.fast_mana.length ? P.fast_mana.join(", ") : "none" },
      { ok: P.extra_turns.length === 0, t: "No chained extra turns", d: P.extra_turns.length ? P.extra_turns.join(", ") : "none" },

    const pushRow = (k, items) => <div className="ic-push" key={k}><span className="ic-push__k">{k}</span><span className="ic-push__v">{items.length ? items.join(", ") : <em>none</em>}</span></div>;
    return (
      <div className="ic-col">
        <div className="ic-eyebrow"><Ic.Crown />Power bracket</div>
        <BracketMeter bracket={b} />
        <hr className="ic-sep" />
        <div className="ic-h">Classification criteria</div>
        <div className="ic-checks">

    const MB = window.MB;

    const passes = [
      { t: "Card count", d: A.stats.total_cards + " / 100 — 99 + commander" },
      { t: "Singleton", d: "No duplicate nonland names" },
      { t: "Commander legality", d: MB.commander.name + " is a legal commander" },
    ];
    return (
      <div className="ic-col">
        <div className="ic-eyebrow" style={{ color: "var(--danger)" }}><Ic.Alert />Validation<span className="ic-eyebrow__c">{errors.length} error{errors.length === 1 ? "" : "s"}</span></div>
        {errors.map((e, i) => (
          <ValidationItem key={i} rule={e.rule} severity={e.severity} card={e.card} detail={e.detail} fixHint="Ask the Oracle to fix legality" />
        ))}
        <hr className="ic-sep" />
        <div className="ic-h">Passing checks</div>
        <div className="ic-checks">
          {passes.map((c, i) => (

              <span className="ic-check__mark ic-check__mark--ok"><Ic.Check /></span>
              <span className="ic-check__body"><div className="ic-check__t">{c.t}</div><div className="ic-check__d">{c.d}</div></span>
            </div>
          ))}
        </div>
        <div className="ic-callout"><Ic.Info /><span>One identity violation blocks legality. Everything else validates — resolve the off-color card and the deck is tournament-legal.</span></div>
      </div>
    );
  }
