/* Manabase UI kit — Collection screen.
   The Collection object (owned cards). NOTE: the engine spec lists collection
   awareness as explicitly out of v1 scope (§12) — this screen is a forward-
   looking mock and is flagged as such in the UI. Grid + table views, a value
   summary, live search and sort. */
(function () {
  const DS = window.ManabaseDesignSystem_9d5890 || {};
  const { CardTile, CardRow, ManaCost, ColorIdentity, Badge, Input, Select, SegmentedControl, Button, IconButton } = DS;
  const Ic = window.Ic;

  const CSS = `
  .col-stats { display:grid; grid-template-columns:repeat(auto-fit, minmax(150px, 1fr)); gap:var(--space-3); margin:0 var(--space-5) var(--space-4); }
  .col-stat { background:var(--surface-card); border:1px solid var(--border-subtle); border-radius:var(--radius-md); padding:var(--space-3) var(--space-4); box-shadow:var(--shadow-bevel); display:flex; flex-direction:column; gap:var(--space-1_5); min-width:0; }
  .col-stat b { font-family:var(--font-mono); font-weight:600; font-size:22px; color:var(--text-primary); line-height:1; font-variant-numeric:tabular-nums; white-space:nowrap; overflow:hidden; text-overflow:ellipsis; }
  .col-stat b.accent { color:var(--accent-text); }
  .col-stat span { font:var(--type-eyebrow); letter-spacing:var(--tracking-caps); text-transform:uppercase; color:var(--text-faint); }
  .col-tile { position:relative; transition:transform var(--dur-fast) var(--ease-out); }
  .col-tile:hover { transform:translateY(-5px); z-index:2; }
  .col-qty { position:absolute; top:8px; left:8px; padding:2px 8px; border-radius:var(--radius-pill); background:color-mix(in oklab, #07080b 72%, transparent); color:var(--text-primary); font:var(--type-data-sm); font-variant-numeric:tabular-nums; border:1px solid var(--border-strong); backdrop-filter:blur(3px); z-index:3; box-shadow:0 2px 6px rgba(0,0,0,.4); }
  .col-empty { padding:var(--space-8); text-align:center; color:var(--text-faint); font:var(--type-body-sm); }
  .col-qtybadge { font-family:var(--font-mono); font-weight:600; font-size:13px; color:var(--accent-text); background:var(--accent-soft); border:1px solid var(--accent-soft-border); border-radius:var(--radius-pill); padding:1px 9px; flex:none; }
  .col-listrow .mb-cardrow__action { opacity:1 !important; transform:none !important; }
  `;
  let injected = false;
  (function ensure(){ if (injected || typeof document==="undefined") return; injected=true; const s=document.createElement("style"); s.setAttribute("data-mb","collection"); s.textContent=CSS; document.head.appendChild(s); })();

  const CI_VAR = { W: "var(--mtg-w)", U: "var(--mtg-u)", B: "var(--mtg-b)", R: "var(--mtg-r)", G: "var(--mtg-g)" };
  const ciColor = (c) => { if (/\bLand\b/.test(c.typeLine)) return "var(--mtg-c)"; const id = c.identity || []; if (!id.length) return "var(--mtg-c)"; if (id.length === 1) return CI_VAR[id[0]] || "var(--mtg-c)"; return "var(--accent)"; };

  const cs = {
    screen: { flex: 1, display: "flex", flexDirection: "column", minHeight: 0 },
    banner: { display: "flex", alignItems: "flex-start", gap: "var(--space-2)", margin: "var(--space-4) var(--space-5) var(--space-4)", padding: "var(--space-2_5) var(--space-3)", background: "var(--info-soft)", border: "1px solid color-mix(in oklab, var(--info) 30%, transparent)", borderRadius: "var(--radius-md)", color: "var(--info)", font: "var(--type-label-sm)", lineHeight: 1.5 },
    toolbar: { display: "flex", alignItems: "center", gap: "var(--space-3)", padding: "0 var(--space-5) var(--space-3)" },
    body: { flex: 1, overflow: "auto", padding: "0 var(--space-5) var(--space-5)" },
    grid: { display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(150px, 1fr))", gap: "var(--space-3)" },
    list: { display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(330px, 1fr))", columnGap: "var(--space-4)", rowGap: "2px" },
  };

  window.Collection = function Collection() {
    const all = window.MB.collection;
    const [view, setView] = React.useState("grid");
    const [query, setQuery] = React.useState("");
    const [sort, setSort] = React.useState("value");

    const total = all.reduce((s, c) => s + c.priceUsd * c.owned, 0);
    const count = all.reduce((s, c) => s + c.owned, 0);
    const top = all.reduce((b, c) => (c.priceUsd * c.owned > (b ? b.priceUsd * b.owned : 0) ? c : b), null);

    const q = query.trim().toLowerCase();
    let items = q ? all.filter((c) => (c.name + " " + c.typeLine).toLowerCase().includes(q)) : all.slice();
    items.sort((a, b) => sort === "name" ? a.name.localeCompare(b.name) : sort === "qty" ? b.owned - a.owned : (b.priceUsd * b.owned) - (a.priceUsd * a.owned));

    return (
      <div style={cs.screen}>
        <div style={cs.banner}>
          <Ic.Info /> Collection awareness is a planned capability — out of v1 engine scope. Shown here as a forward-looking surface.
        </div>

        <div className="col-stats">
          <div className="col-stat"><b className="accent">{count}</b><span>Cards owned</span></div>
          <div className="col-stat"><b>{all.length}</b><span>Unique</span></div>
          <div className="col-stat"><b>${total.toFixed(0)}</b><span>Total value</span></div>
          <div className="col-stat"><b>{top ? top.name.split(",")[0] : "—"}</b><span>Most valuable</span></div>
        </div>

        <div style={cs.toolbar}>
          <div style={{ flex: "0 1 280px", minWidth: 170 }}><Input leadingIcon={<Ic.Search />} placeholder="Search your collection…" value={query} onChange={(e) => setQuery(e.target.value)} /></div>
          <Badge tone="neutral" mono>{items.length} shown</Badge>
          <div style={{ flex: 1 }} />
          <Select size="sm" value={sort} onChange={(e) => setSort(e.target.value)} options={[{ value: "value", label: "Sort: Value" }, { value: "name", label: "Sort: Name" }, { value: "qty", label: "Sort: Quantity" }]} />
          <SegmentedControl value={view} onChange={setView} options={[{ value: "grid", icon: <Ic.Grid />, label: "Grid" }, { value: "table", icon: <Ic.List />, label: "List" }]} />
        </div>

        <div style={cs.body}>
          {!items.length ? <div className="col-empty">No owned cards match “{query}”.</div> : view === "grid" ? (
            <div style={cs.grid}>
              {items.map((c) => (
                <div key={c.name} className="col-tile">
                  <CardTile name={c.name} typeLine={c.typeLine} identity={c.identity} image={c.image} priceUsd={c.priceUsd}
                    cost={<ManaCost cost={c.manaCost} size={12} />}
                    footer={<Button size="sm" fullWidth variant="secondary" leadingIcon={<Ic.Plus />}>Add to deck</Button>} />
                  <span className="col-qty">×{c.owned}</span>
                </div>
              ))}
            </div>
          ) : (
            <div style={cs.list}>
              {items.map((c) => (
                <div key={c.name} className="deckrow col-listrow" style={{ "--ci": ciColor(c) }}>
                  <img className="deckrow__art" src={c.image} alt="" loading="lazy" aria-hidden="true" onLoad={(e) => { e.currentTarget.classList.add("loaded"); }} onError={(e) => { e.currentTarget.style.display = "none"; }} />
                  <span className="deckrow__veil" />
                  <CardRow name={c.name} typeLine={c.typeLine} identity={c.identity} cost={<ManaCost cost={c.manaCost} size={13} />}
                    priceUsd={c.priceUsd}
                    action={<span className="col-qtybadge">×{c.owned}</span>} />
                </div>
              ))}
            </div>
          )}
        </div>
      </div>
    );
  };
})();
