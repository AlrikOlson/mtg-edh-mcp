  .ins-whatif { display:flex; align-items:center; gap:var(--space-2); margin:var(--space-3) var(--space-4) 0; padding:var(--space-2) var(--space-3); background:color-mix(in oklab,var(--gold-400) 9%,var(--surface-card)); border:1px dashed var(--accent-soft-border); border-radius:var(--radius-md); box-shadow:var(--shadow-bevel); animation:cap-in var(--dur-base) var(--ease-out); }

  .ghostfill { position:absolute; top:-1px; bottom:-1px; left:0; border-radius:var(--radius-pill); border:1.5px dashed color-mix(in oklab,var(--accent-text) 85%,transparent); background:color-mix(in oklab,var(--gold-400) 14%,transparent); box-sizing:border-box; transition:width var(--dur-slow) var(--ease-out); pointer-events:none; z-index:2; }
  .ins-pip__val--whatif, .ins-comp__row--whatif > span:last-child { display:inline-flex; align-items:center; justify-content:flex-end; gap:4px; }
  `;
  let injected = false;
  (function ensure(){ if (injected || typeof document==="undefined") return; injected=true; const s=document.createElement("style"); s.setAttribute("data-mb","insights"); s.textContent=CSS; document.head.appendChild(s); })();

  window.InsightsRail = function InsightsRail({ analysis, projected, hasIllegal, deck }) {
    const A = analysis;
    const PR = projected;
    const whatif = !!PR;
    const D = deck || [];
    const NAV_ICON = { stats: <Ic.Info />, curve: <Ic.Graph />, pips: <Ic.Percent />, comp: <Ic.Box />, mana: <Ic.Flask />, roles: <Ic.Sparkles />, goldfish: <Ic.Dice />, bracket: <Ic.Crown />, valid: <Ic.Alert /> };

    const bodyRef = React.useRef(null);
    const refs = React.useRef({});
    const [active, setActive] = React.useState("stats");

    const sections = [
      { key: "stats", label: "At a glance", card: true, node: <IC.GlanceCard A={A} PR={PR} deck={D} /> },
      { key: "curve", label: "Mana curve", card: true, node: <IC.CurveCard A={A} PR={PR} deck={D} /> },
      { key: "pips", label: "Color pips", card: true, node: <IC.PipCard A={A} PR={PR} /> },
      { key: "comp", label: "Composition", card: true, node: <IC.TypeCard A={A} PR={PR} deck={D} /> },
      { key: "mana", label: "Mana sources", card: true, node: <IC.SourceCard A={A} /> },
      { key: "roles", label: "Role coverage", card: true, node: <IC.RoleCard A={A} PR={PR} deck={D} /> },
      { key: "goldfish", label: "Goldfish", card: true, node: <IC.GoldfishCard A={A} PR={PR} /> },
      { key: "bracket", label: "Power bracket", card: true, node: <IC.BracketCard A={A} /> },
    ];
    if (hasIllegal) sections.push({ key: "valid", label: "Validation", card: true, danger: true, node: <IC.ValidationCard A={A} /> });

    const navBtns = React.useRef({});
    const lock = React.useRef(false);
    const target = React.useRef(null);
    const lockTimer = React.useRef(0);
    const spacerRef = React.useRef(null);
    const [thumbTop, setThumbTop] = React.useState(0);
    const [track, setTrack] = React.useState({ top: 0, height: 0 });
    const [progress, setProgress] = React.useState({ top: 0, height: 0 });
    const activeLabel = (sections.find((s) => s.key === active) || sections[0]).label;
    const LINE = 44; // detection line, px below the body's top edge

    // Scroll-spy in VIEWPORT coordinates (robust to offsetParent), picking the
    // last section whose top has crossed the detection line.
    const spyNow = () => {
      const body = bodyRef.current; if (!body) return active;
      const line = body.getBoundingClientRect().top + LINE;
      let cur = sections[0].key;
      for (const s of sections) { const el = refs.current[s.key]; if (!el) continue; if (el.getBoundingClientRect().top - 2 <= line) cur = s.key; else break; }
      if (body.scrollTop + body.clientHeight >= body.scrollHeight - 4) cur = sections[sections.length - 1].key;
      return cur;
    };
    const releaseLock = () => { lock.current = false; };
    const onScroll = () => {
      const body = bodyRef.current; if (!body) return;
      if (lock.current) {
        // hold the spy until the programmatic scroll actually arrives (no
        // mid-flight drift); a fallback timer / user gesture also releases it
        if (target.current == null || Math.abs(body.scrollTop - target.current) <= 2) releaseLock();
        else return;
      }
      const cur = spyNow();
      if (cur !== active) setActive(cur);
    };
    const jump = (key) => {
      const el = refs.current[key], body = bodyRef.current; if (!el || !body) return;
      const top = body.scrollTop + (el.getBoundingClientRect().top - body.getBoundingClientRect().top) - LINE + 2;
      const clamped = Math.max(0, Math.min(Math.round(top), body.scrollHeight - body.clientHeight));
      target.current = clamped; lock.current = true; setActive(key);
      body.scrollTo({ top: clamped, behavior: "smooth" });
      clearTimeout(lockTimer.current); lockTimer.current = setTimeout(releaseLock, 1000);
    };
    const center = (b) => b.offsetTop + b.offsetHeight / 2;

    // A trailing spacer sized so EVERY section (incl. the last) can scroll up to
    // the detection line — keeps the click target and the spy in agreement.
    React.useEffect(() => {
      const body = bodyRef.current, sp = spacerRef.current; if (!body || !sp) return;
      const fit = () => { const keys = Object.keys(refs.current); const last = refs.current[keys[keys.length - 1]]; if (last) sp.style.height = Math.max(0, body.clientHeight - last.offsetHeight - LINE - 12) + "px"; };
      fit(); const t = setTimeout(fit, 140); window.addEventListener("resize", fit);
      return () => { window.removeEventListener("resize", fit); clearTimeout(t); };
    }, [analysis, projected, hasIllegal, whatif]);

    // a manual scroll gesture cancels the programmatic lock immediately
    React.useEffect(() => {
      const body = bodyRef.current; if (!body) return;
      body.addEventListener("wheel", releaseLock, { passive: true });
      body.addEventListener("touchstart", releaseLock, { passive: true });
      return () => { body.removeEventListener("wheel", releaseLock); body.removeEventListener("touchstart", releaseLock); };
    }, []);

    React.useEffect(() => {
      const measure = () => {
        const b = navBtns.current[active];
        const first = navBtns.current[sections[0].key];
        const last = navBtns.current[sections[sections.length - 1].key];
        if (b) setThumbTop(b.offsetTop);
        if (first && last) setTrack({ top: center(first), height: Math.max(0, center(last) - center(first)) });
        if (b && first) setProgress({ top: center(first), height: Math.max(0, center(b) - center(first)) });
      };
      measure();
      // re-measure after fonts/icons settle and on resize (layout shifts move offsetTop)
      const t = setTimeout(measure, 380);
      window.addEventListener("resize", measure);
      return () => { clearTimeout(t); window.removeEventListener("resize", measure); };
    }, [active, hasIllegal, whatif]);

    return (
      <div className="ins-rail">
        <div className="ins-head">
          <span className="ins-head__t">Insights</span>
          <span className="ins-head__sep" />
          <span className="ins-head__active">{activeLabel}</span>
          <div style={{ flex: 1 }} />
          <Badge tone={hasIllegal ? "danger" : "success"} dot>{hasIllegal ? "1 error" : "Legal"}</Badge>
          <Badge tone="accent" className="ins-head__bracket">{"Bracket " + A.bracket.bracket}</Badge>
        </div>
        {whatif ? (
          <div className="ins-whatif">
            <Ic.Compare />
            <span className="ins-whatif__t"><b>What-if preview</b> · projected impact</span>
            <span className="ins-whatif__leg">
              <span className="ins-whatif__sw"><i></i>Now</span>
              <span className="ins-whatif__sw"><i className="ghost"></i>If accepted</span>
            </span>
          </div>
        ) : null}
        <div className="ins-main">
          <div className="ins-subnav">
            <div className="ins-subnav__track" style={{ top: track.top, height: track.height }} />
            <div className="ins-subnav__progress" style={{ top: progress.top, height: progress.height }} />
            <div className="ins-subnav__thumb" style={{ top: thumbTop }} />
            {sections.map((s) => (
              <button key={s.key} ref={(el) => (navBtns.current[s.key] = el)} className={"ins-subnav__item" + (active === s.key ? " ins-subnav__item--active" : "") + (s.danger ? " ins-subnav__item--danger" : "")} data-label={s.label} onClick={() => jump(s.key)} aria-label={s.label} aria-current={active === s.key ? "true" : undefined}>{NAV_ICON[s.key] || <Ic.Info />}</button>
            ))}
          </div>
          <div className="ins-body" ref={bodyRef} onScroll={onScroll}>
            {sections.map((s) => (
              <div key={s.key} ref={(el) => (refs.current[s.key] = el)} className={(s.card ? "ins-card" : "") + (s.card && active === s.key ? " ins-card--hero" : "")} style={s.danger ? { borderColor: "color-mix(in oklab, var(--danger) 35%, transparent)" } : null}>
                {s.node}
              </div>
            ))}
            <div ref={spacerRef} aria-hidden="true" style={{ flex: "none" }} />
          </div>
        </div>
      </div>
    );
  };
})();
