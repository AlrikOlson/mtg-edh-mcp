                onChange={(e) => setNl(e.target.value)}
                onKeyDown={(e) => { if (e.key === "Enter" && nl.trim()) { const m = SUGG.find((s) => nl.toLowerCase().includes(s.id) || s.label.toLowerCase().includes(nl.toLowerCase().slice(0, 5))); runSearch(m ? m.id : "proliferate", nl); } }} />
              {thinking ? <span style={{ font: "var(--type-label-sm)", color: "var(--accent-text)" }}>Searching…</span> : <Ic.Sparkles />}
            </div>
            {phase === "idle" ? (
              <div className="cb-nlchips">
                {SUGG.map((s) => <button key={s.id} className="cb-chip" onClick={() => runSearch(s.id)}><Ic.Sparkles />{s.label}</button>)}
              </div>
            ) : (
              <div className="cb-cap"><span className="cb-cap__tool mb-mono">card_search</span><span className="cb-cap__args mb-mono">{active && phase !== "idle" ? (cap ? cap.args : active.query) : ""}</span><span className="cb-cap__res mb-mono">{cap ? (cap.res === "…" ? "…" : "→ " + cap.res) : ""}</span></div>
            )}
            {/* translated scryfall query + facet tags */}
            <Input mono leadingIcon={<Ic.Search />} value={active.query} readOnly />
            <div style={{ display: "flex", gap: "var(--space-1_5)", flexWrap: "wrap" }}>
              {active.tags.map((t, i) => <Tag key={i} tagKey={t[0]} accent={!!t[2]} onRemove={() => {}}>{t[1]}</Tag>)}
            </div>
          </div>
          <div style={bs.toolbar}>
            <span style={{ font: "var(--type-label)", color: "var(--text-secondary)", whiteSpace: "nowrap" }}>
              <b style={{ color: "var(--text-primary)", fontFamily: "var(--font-mono)" }}>{active.count}</b> results
            </span>
            <div style={{ flex: 1 }} />
            <div style={{ width: 160, flex: "none" }}><Select size="sm" options={["EDHREC rank", "Mana value", "Price", "Name"]} /></div>
            <SegmentedControl value={view} onChange={setView} options={[{ value: "table", icon: <Ic.List />, label: "Table" }, { value: "grid", icon: <Ic.Grid />, label: "Grid" }]} />
          </div>
          {thinking ? (
            <div style={bs.results}>{[0, 1, 2, 3, 4, 5].map((i) => <div key={i} className="cb-skel" style={{ opacity: 1 - i * 0.13 }} />)}</div>
          ) : view === "table" ? (
            <div style={bs.results} key={"t" + searchId}>
              {results.map((c, i) => (
                <div key={c.name} className="cb-row" style={{ animationDelay: i * 35 + "ms" }}>
                  <CardRow name={c.name} typeLine={c.typeLine} identity={c.identity} image={c.image}
                    cost={<ManaCost cost={c.manaCost} size={13} />}
                    priceUsd={c.priceUsd} edhrecRank={c.edhrecRank} selected={selected && selected.name === c.name}
                    onClick={() => setSelected(c)} action={addBtn()} />
                </div>
              ))}
            </div>
          ) : (
            <div className="cb-gal" key={"g" + searchId}>
              {results.map((c, i) => (
                <GalleryCard key={c.name} c={c} idx={i} selected={selected && selected.name === c.name} onSelect={setSelected} />
              ))}
            </div>
          )}
        </div>
        <div style={bs.inspector}><Inspector card={selected} onAdd={() => {}} /></div>
      </div>
    );
  };
})();
